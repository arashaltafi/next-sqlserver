CREATE PROCEDURE [dbo].[Product_Create]
    @Name NVARCHAR(150),
    @Description NVARCHAR(500),
    @Price DECIMAL(18,2),
    @Stock INT
AS
BEGIN

    SET NOCOUNT ON;

    INSERT INTO dbo.Products
    (
        Name,
        Description,
        Price,
        Stock
    )
    VALUES
    (
        @Name,
        @Description,
        @Price,
        @Stock
    );

    SELECT
        SCOPE_IDENTITY() AS Id;
END;
GO

GO

CREATE PROCEDURE [dbo].[Product_Update]
    @Id INT,
    @Name NVARCHAR(150),
    @Description NVARCHAR(500),
    @Price DECIMAL(18,2),
    @Stock INT
AS
BEGIN

    SET NOCOUNT ON;

    UPDATE dbo.Products
    SET
        Name = @Name,
        Description = @Description,
        Price = @Price,
        Stock = @Stock,
        UpdatedAt = GETDATE()
    WHERE Id = @Id;

    SELECT
        Id,
        Name,
        Description,
        Price,
        Stock,
        CreatedAt,
        UpdatedAt
    FROM dbo.Products
    WHERE Id = @Id;

END;
